   5: controlled RTT + loss distributions and verifies that the BE-RAR
   6: scorer (driving `_pick_best_relay` when ``ol_coherence_field`` is
   7: available) makes physically sensible picks.
   8: 
   9: The BE-RAR shape ``nu(y) = 1/(1 − exp(−√y))`` is the same function

  78: def _expected_best_be_rar(metrics_per_url: dict[str, dict]) -> str:
  79:     """The ground-truth best relay UNDER BE-RAR's own cost function.
  80:     Computes ``rtt × nu(y_quality)`` where ``y_quality = (1-loss)/loss``,
  81:     independently from the daemon's picker, and returns the url that
  82:     minimizes it. Tests assert the picker agrees with this."""
  83:     import math
  84: 
  85:     def nu(y: float) -> float:
  86:         if y <= 0:
  87:             return float("inf")
  88:         return 1.0 / (1.0 - math.exp(-math.sqrt(y)))
  89: 
  90:     best_url = None
  91:     best_cost = float("inf")
  92:     for url, m in metrics_per_url.items():
  93:         loss = min(max(m["loss_rate"], 1e-6), 1.0 - 1e-6)
  94:         y_quality = (1.0 - loss) / loss
  95:         cost = m["rtt_ms"] * nu(y_quality)
  96:         if cost < best_cost:
  97:             best_url = url
  98:             best_cost = cost

 108:     differences."""

 263: def test_be_rar_softer_asymptote_vs_heuristic_on_moderate_loss():
 264:     """Compute BE-RAR penalty and heuristic penalty at the same loss
 265:     levels and verify BE-RAR is SOFTER (smaller penalty) at moderate

 268:     import math

 270: 
 271:     def heuristic(loss: float) -> float:
 272:         return 1.0 / max(1.0 - loss, 1e-9) ** 2
 273: 
 274:     def be_rar_penalty(loss: float) -> float:
 275:         loss = min(max(loss, 1e-6), 1.0 - 1e-6)
 276:         y = (1.0 - loss) / loss
